// apps/cli/contract.mjs
// The bibrain-agent/1 contract as the CLI speaks it. `hook`, `hook-end`, `recall` and `note` are four parsers and printers over the two functions here.
import { readFileSync } from "node:fs";
import { HOOK_BUDGET_MS, HOOK_END_BUDGET_MS, HOOK_END_PER_SESSION, HOOK_PROMPT_MAX, firstCodePoints, hasMarkerLine, hookSkipReason, hookStatePath, keyFilePath, looksLikeSecret, loopbackApi, markerLines, noteHookState, noteSessionEnd, readHookState, sha256, writeHookState } from "./lib.mjs";

export const CONTRACT = "bibrain-agent/1";
export const GENERIC_CLIENTS = ["hermes", "other"];

/** The client name a generic command reports: one of the generic names, else other. */
export function genericClient(flag) {
  return typeof flag === "string" && GENERIC_CLIENTS.includes(flag) ? flag : "other";
}

function parseV1(raw) {
  let value;
  try {
    value = JSON.parse(String(raw).replace(/^\uFEFF/u, "").trim());
  } catch {
    return null;
  }
  if (!value || typeof value !== "object" || value.v !== 1) return null;
  return value;
}

/** Generic recall input: { v: 1, prompt, session?, cwd? }. The cwd is accepted and ignored; it never leaves the process. */
export function parseRecallInput(raw) {
  const value = parseV1(raw);
  if (!value || typeof value.prompt !== "string") return null;
  return { prompt: value.prompt, session: typeof value.session === "string" ? value.session : "" };
}

/** Generic note input: { v: 1, reply, session?, continued? }. */
export function parseNoteInput(raw) {
  const value = parseV1(raw);
  if (!value || typeof value.reply !== "string") return null;
  return { reply: value.reply, session: typeof value.session === "string" ? value.session : "", continued: value.continued === true };
}

export const recallOutput = (text) => JSON.stringify({ v: 1, context: text });
export const noteOutput = () => JSON.stringify({ v: 1 });

function connection(dataDir, connectionId) {
  const file = keyFilePath(dataDir, connectionId);
  const { api, key } = JSON.parse(readFileSync(file, "utf8"));
  if (typeof key !== "string") throw new Error("no key");
  return { origin: loopbackApi(api), key, statePath: hookStatePath(dataDir, connectionId) };
}

/**
 * One prompt-time recall push. Returns the context text to put in front of the model, or "" when there is nothing.
 * Throws on any problem; every caller swallows that, because a hook must never slow or block the agent.
 */
export async function pushRecall({ prompt, sessionId, client, dataDir, connectionId, startedAt }) {
  if (hookSkipReason(prompt)) return "";
  const { origin, key, statePath } = connection(dataDir, connectionId);
  const state = readHookState(statePath);
  const sessionHash = sha256(sessionId);
  const session = state.sessions[sessionHash];
  // performance.timeOrigin is fractional and AbortSignal.timeout needs an integer.
  const remaining = Math.max(50, Math.floor(HOOK_BUDGET_MS - (Date.now() - startedAt)));
  const res = await fetch(`${origin}/api/v1/agent/recall-push`, {
    method: "POST",
    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
    body: JSON.stringify({ prompt: firstCodePoints(prompt, HOOK_PROMPT_MAX), session_id: sessionHash, first_prompt: !session?.digested, seen: session?.hashes ?? [], client }),
    signal: AbortSignal.timeout(remaining),
    redirect: "error",
  });
  if (!res.ok) return "";
  const body = await res.json();
  if (!body || typeof body !== "object" || !["full", "pointer", "none"].includes(body.level) || typeof body.text !== "string") return "";
  const ids = Array.isArray(body.hit_ids) ? body.hit_ids.filter((x) => typeof x === "string") : [];
  writeHookState(statePath, noteHookState(state, sessionHash, ids, Date.now(), body.digest === true));
  return body.text;
}

/** One end-of-turn session note: only the marker lines travel, once per session and text, at most 12 a session. Throws on any problem; callers swallow it. */
export async function postNote({ message, active, sessionId, client, dataDir, connectionId, startedAt }) {
  if (active || !hasMarkerLine(message) || looksLikeSecret(message)) return;
  const text = markerLines(message);
  if (text === null || looksLikeSecret(text)) return;
  const { origin, key, statePath } = connection(dataDir, connectionId);
  const state = readHookState(statePath);
  // A client that sends no session id shares one bucket (and its per-session note cap) across those turns.
  const sessionHash = sha256(sessionId);
  const messageHash = sha256(text);
  const notes = state.sessions[sessionHash]?.notes ?? [];
  if (notes.includes(messageHash) || notes.length >= HOOK_END_PER_SESSION) return;
  const remaining = Math.max(50, Math.floor(HOOK_END_BUDGET_MS - (Date.now() - startedAt)));
  const res = await fetch(`${origin}/api/v1/agent/session-note`, {
    method: "POST",
    headers: { Authorization: `Bearer ${key}`, "Content-Type": "application/json" },
    body: JSON.stringify({ session_id: sessionHash, client, text }),
    signal: AbortSignal.timeout(remaining),
    redirect: "error",
  });
  if (!res.ok) return;
  writeHookState(statePath, noteSessionEnd(state, sessionHash, messageHash, Date.now()));
}
```

Order of checks in `pushRecall` is the order `hook()` had: skip reason, key file, loopback check, state, fetch. Keep it so the hook's "makes no call for skips" and "missing key file" timings stay identical.

// A hook must never slow or block the agent: any problem ends silently with exit 0.
// Exit code 2 would block the prompt, so nothing here can produce it.
async function hook() {
  try {
    const client = String(positional[0] ?? "");
    if (!HOOK_CLIENTS.includes(client)) return;
    const raw = await readStdin(HOOK_STDIN_MS);
    if (raw === null) return;
    const event = parseHookStdin(raw);
    if (!event) return;
    const text = await pushRecall({ prompt: event.prompt, sessionId: event.session_id, client, dataDir: String(flags.data ?? ""), connectionId: String(flags.connection ?? ""), startedAt });
    if (text) process.stdout.write(hookOutput(text));
  } catch {
    /* silent by design */
  }
  process.exitCode = 0;
}

// The end-of-turn hook: it may add nothing and block nothing. Any problem ends with "{}" and exit 0.
async function hookEnd() {
  try {
    const client = String(positional[0] ?? "");
    if (!HOOK_CLIENTS.includes(client)) return;
    const raw = await readStdin(HOOK_STDIN_MS);
    if (raw === null) return;
    const event = parseStopStdin(raw);
    if (!event) return;
    await postNote({ message: event.message, active: event.active, sessionId: event.session_id, client, dataDir: String(flags.data ?? ""), connectionId: String(flags.connection ?? ""), startedAt });
  } catch {
    /* silent by design */
  } finally {
    process.stdout.write(hookEndOutput());
    process.exitCode = 0;
  }
}

// The generic contract (bibrain-agent/1): { v: 1, prompt, session } in, { v: 1, context } out, always, exit 0 always.
async function recall() {
  let text = "";
  try {
    const raw = await readStdin(HOOK_STDIN_MS);
    const input = raw === null ? null : parseRecallInput(raw);
    if (input) text = await pushRecall({ prompt: input.prompt, sessionId: input.session, client: genericClient(flags.client), dataDir: String(flags.data ?? ""), connectionId: String(flags.connection ?? ""), startedAt });
  } catch {
    text = "";
  } finally {
    process.stdout.write(recallOutput(text));
    process.exitCode = 0;
  }
}

async function note() {
  try {

import { TOOL_DESCRIPTIONS, TOOL_NAMES, recallOutputShape, renderRecallContent, type ToolName } from "./surface.ts";

/** One call to the BiBrain HTTP API as the connection: GET when there is no body, POST with an Idempotency-Key otherwise. Throws with a plain message on any failure. */
export type Call = (path: string, body?: unknown, idempotencyKey?: string) => Promise<any>;

const project = z.string().min(1).max(100);
const object = z.string().min(1).max(100);
const query = (path: string, values: Record<string, string | number | boolean | undefined>) =>
  path + "?" + new URLSearchParams(Object.entries(values).filter(([, v]) => v !== undefined).map(([k, v]) => [k, String(v)])).toString();
const READ_ONLY = new Set<ToolName>(["brain_recall", "brain_open"]);

/** The seven tools, exactly as the stdio adapter has registered them since plan 5. Both transports register from this table, so they can never drift. */
export const TOOL_TABLE: Record<ToolName, { schema: z.ZodRawShape; execute: (call: Call, v: any) => Promise<any> }> = {
  brain_recall: { schema: { query: z.string().min(1).max(500), project_id: project.optional(), history: z.boolean().optional(), limit: z.number().int().min(1).max(20).optional() }, execute: (call, v) => call(query("/agent/recall", v)) },
  brain_open: { schema: { id: object, version: z.number().int().positive().optional(), compare_version: z.number().int().positive().optional() }, execute: (call, v) => call(query("/agent/open", v)) },
  // brain_propose, brain_note, brain_forget, brain_link, brain_intent: the exact schemas from main.ts, with `execute: (call, v) => …` in place of the closure over the module-level call.
};

export function registerBrainTools(server: McpServer, call: Call, allowed: Set<string>): void {
  for (const name of TOOL_NAMES) {
    if (!allowed.has(name)) continue;
    const config = TOOL_TABLE[name];
    server.registerTool(
      name,
      {
        description: TOOL_DESCRIPTIONS[name],
        inputSchema: config.schema,
        ...(name === "brain_recall" ? { outputSchema: recallOutputShape } : {}),
        annotations: { readOnlyHint: READ_ONLY.has(name), destructiveHint: name === "brain_forget", idempotentHint: READ_ONLY.has(name), openWorldHint: false },
      },
      async (input: any) => {
        try {
          const value = await config.execute(call, input);
          await call("/agent/verify", {});
          if (name === "brain_recall") return renderRecallContent(value);
          return { content: [{ type: "text" as const, text: JSON.stringify(value) }] };
        } catch (error) {
          return { isError: true, content: [{ type: "text" as const, text: error instanceof Error ? error.message : "Operation failed." }] };
        }
      },
    );
  }
}

 * Streamable HTTP MCP, stateless and JSON-only, for harnesses that only speak remote MCP. Every tool call goes back through this
 * server's own HTTP routes with the caller's bearer, so authorization, validation, idempotency and the content-free logs are the routes'
 * own and the same as the stdio adapter's. There is no session, no stream and no OAuth: bearer grants only, behind the request hook
 * that already checks Host, Origin, path escapes and navigations.
 */
export function mcpHttpRoute(app: FastifyInstance) {
  const refuse = async (_req: FastifyRequest, reply: FastifyReply) => {
    reply.code(405).header("Allow", "POST").send({ error: { code: "INVALID_INPUT", message: NOT_POST, request_id: _req.id, retryable: false } });
  };
  app.get("/api/v1/agent/mcp", refuse);
  app.delete("/api/v1/agent/mcp", refuse);
  app.post("/api/v1/agent/mcp", { bodyLimit: MCP_HTTP_BODY_LIMIT, config: { rateLimit: { max: 120, timeWindow: 60000 } } }, async (req, reply) => {
    if (req.principal.kind !== "agent") throw new BrainError("FORBIDDEN", "Use an agent token.", 403);
    const authorization = req.headers.authorization!;
    const host = req.headers.host ?? "";
    // The same door the stdio adapter uses: one request per call, same bearer, same Host, no cookie.
    const call: Call = async (path, body, idempotencyKey) => {
      const res = await app.inject({
        method: body === undefined ? "GET" : "POST",
        url: "/api/v1" + path,
        headers: { host, authorization, "content-type": "application/json", ...(body === undefined ? {} : { "idempotency-key": idempotencyKey ?? crypto.randomUUID() }) },
        ...(body === undefined ? {} : { payload: body }),
      });
      const value = res.json();
      if (res.statusCode >= 400) throw new Error(value?.error?.message ?? "Access unavailable.");
      return value;
    };
    const capabilities = await call("/agent/capabilities");
    const server = new McpServer({ name: "bi-brain", version: "0.3.0" }, { instructions: SERVER_INSTRUCTIONS });
    registerBrainTools(server, call, new Set<string>(capabilities.tools));
    server.registerResource("capabilities", "brain://capabilities", { mimeType: "application/json" }, async () => ({
      contents: [{ uri: "brain://capabilities", text: JSON.stringify({ tools: capabilities.tools, projects: capabilities.projects ?? [], surface: "v2", contract: AGENT_CONTRACT, transport: "streamable-http", private_sources: "owner only" }) }],
    }));
    const transport = new WebStandardStreamableHTTPServerTransport({ sessionIdGenerator: undefined, enableJsonResponse: true });
    await server.connect(transport);
    try {
      const request = new Request(`http://${host}/api/v1/agent/mcp`, { method: "POST", headers: { accept: String(req.headers.accept ?? ""), "content-type": "application/json" }, body: JSON.stringify(req.body) });
      const res = await transport.handleRequest(request, { parsedBody: req.body });
      const text = await res.text();
      reply.code(res.status);
      const type = res.headers.get("content-type");
      if (type) reply.header("content-type", type);
      return reply.send(text);
    } finally {
      await transport.close().catch(() => {});
    }
  });
}

"""BiBrain for Hermes Agent: recall in front of each turn, a session note after it (bibrain-agent/1).

Fail-open by design: nothing here raises out of a hook, blocks a turn, or logs the key, the prompt, the reply or the returned text.

import hashlib
import json
import logging
import os
import re
import urllib.error
import urllib.request
from collections import OrderedDict

log = logging.getLogger(__name__)

CLIENT = "hermes"

MIN_PROMPT = 12
RECALL_TIMEOUT_S = 1.5
NOTE_TIMEOUT_S = 2.5
SESSIONS = 20
HASHES = 300
NOTES_PER_SESSION = 12
NOTE_MAX = 4000
LINE_MAX = 300
DECISIONS = 10
QUESTIONS = 10

# The same three patterns as apps/cli/lib.mjs looksLikeSecret; the server checks again.
_SECRETS = [
    re.compile(r"-----BEGIN (?:RSA |EC |OPENSSH |DSA )?PRIVATE KEY-----"),
    re.compile(r"\b(?:sk-(?:proj-)?[A-Za-z0-9_-]{20,}|gh[pousr]_[A-Za-z0-9]{20,}|github_pat_[A-Za-z0-9_]{20,}|AKIA[A-Z0-9]{16})\b"),
    re.compile(r"(?:api[_ -]?key|access[_ -]?token|client[_ -]?secret|password)\s*[=:]\s*[\"']?[A-Za-z0-9_./+\-=]{16,}", re.I),
]
_MARKER = re.compile(r"^\s*(?:[-*•]\s*)?(?:\*\*)?(?:decided|open question|still unclear|question|unclear|next step)\s*:", re.I | re.M)


def looks_like_secret(text):
    return any(p.search(text) for p in _SECRETS)


def skip_reason(prompt):
    t = prompt.strip()
    if len(t) < MIN_PROMPT:
        return "short"
    if t[:1] in "/!#":
        return "command"
    if looks_like_secret(t):
        return "secret"
    return None


def api_origin(value):
    """Only https, or plaintext loopback with a port; no credentials, path, query or fragment. The same rule as the stdio adapter."""
    from urllib.parse import urlsplit
    u = urlsplit(value or "")
    local = u.scheme == "http" and u.hostname in ("127.0.0.1", "localhost") and u.port
    if (not local and u.scheme != "https") or u.username or u.password or u.path not in ("", "/") or u.query or u.fragment:
        raise ValueError("BRAIN_API_URL must be an HTTPS origin, or loopback HTTP with a port, without credentials or a path.")
    return f"{u.scheme}://{u.netloc}"


def marker_lines(text):
    """Only the marker lines of a reply, cleaned as the server cleans them: the last 10 decisions, the last 10 questions and the last next step,
    each at most 300 code points, joined by newlines and cut to 4,000. None unless a decision or a question is present."""
    decisions, questions, nxt, fenced = [], [], None, False
    fits = lambda s: 0 < len(s) <= LINE_MAX
    for raw in re.split(r"\r?\n", text):
        if re.match(r"^\s*```", raw):
            fenced = not fenced
            continue
        if fenced:
            continue
        line = re.sub(r"^[-*•]\s*", "", raw.strip()).replace("**", "").strip()
        d = re.match(r"^decided\s*:\s*(.+)$", line, re.I)
        if d:
            m = re.match(r"^([^:\r\n]{1,120}):\s+(.+)$", d.group(1).strip())
            if m and fits(m.group(2).strip()):
                decisions.append(line)
            continue
        q = re.match(r"^(?:open question|still unclear|question|unclear)\s*:\s*(.+)$", line, re.I)
        if q:
            if fits(q.group(1).strip()):
                questions.append(line)
            continue
        n = re.match(r"^next step\s*:\s*(.+)$", line, re.I)
        if n and fits(n.group(1).strip()):
            nxt = line
    if not decisions and not questions:
        return None
    return "\n".join(decisions[-DECISIONS:] + questions[-QUESTIONS:] + ([nxt] if nxt else []))[:NOTE_MAX]


def _sha(s):
    return hashlib.sha256(s.encode("utf-8")).hexdigest()


def _settings():
    """The key and the address, from Hermes's secret scope when it exists, else the environment. Never logged."""
    get = None
    try:
        from agent.secret_scope import get_secret as get
    except Exception:
        pass
    read = lambda name: (get(name) if get else None) or os.environ.get(name) or ""
    key, url = read("BRAIN_AGENT_TOKEN").strip(), read("BRAIN_API_URL").strip()
    if not key or not url:
        return None
    return api_origin(url), key


def _post(path, body, timeout):
    origin, key = _settings()
    req = urllib.request.Request(origin + path, data=json.dumps(body).encode("utf-8"), method="POST", headers={"Authorization": "Bearer " + key, "Content-Type": "application/json"})
    # No redirects: an opener without the redirect handler refuses them.
    opener = urllib.request.build_opener(_NoRedirect())
    with opener.open(req, timeout=timeout) as res:
        return json.loads(res.read().decode("utf-8"))


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, *a, **k):
        return None


# Per-session memory, in this process only: hashes of ids already shown, whether the digest was carried, hashes of notes sent.
_sessions = OrderedDict()


def _session(session_hash):
    s = _sessions.pop(session_hash, None) or {"hashes": [], "digested": False, "notes": []}
    _sessions[session_hash] = s
    while len(_sessions) > SESSIONS:
        _sessions.popitem(last=False)
    return s


def pre_llm_call(**kwargs):
    try:
        prompt = kwargs.get("user_message")
        if not isinstance(prompt, str) or skip_reason(prompt):
            return None
        if not _settings():
            return None
        session_hash = _sha(str(kwargs.get("session_id") or ""))
        s = _session(session_hash)
        body = {"prompt": prompt[:PROMPT_MAX], "session_id": session_hash, "first_prompt": not s["digested"], "seen": s["hashes"][-HASHES:], "client": CLIENT}
        reply = _post("/api/v1/agent/recall-push", body, RECALL_TIMEOUT_S)
        if not isinstance(reply, dict) or reply.get("level") not in ("full", "pointer", "none") or not isinstance(reply.get("text"), str):
            return None
        ids = [x for x in (reply.get("hit_ids") or []) if isinstance(x, str)]
        s["hashes"] = (s["hashes"] + [_sha(i) for i in ids])[-HASHES:]
        s["digested"] = s["digested"] or reply.get("digest") is True
        log.debug("bibrain recall level=%s hits=%d", reply.get("level"), len(ids))
        return {"context": reply["text"]} if reply["text"] else None
    except Exception as e:  # fail-open: the turn goes on without the brain
        log.debug("bibrain recall skipped: %s", type(e).__name__)
        return None


def post_llm_call(**kwargs):
    try:
        reply = kwargs.get("assistant_response")
        if not isinstance(reply, str) or not _MARKER.search(reply) or looks_like_secret(reply):
            return None
        text = marker_lines(reply)
        if text is None or looks_like_secret(text):
            return None
        if not _settings():
            return None
        session_hash = _sha(str(kwargs.get("session_id") or ""))
        s = _session(session_hash)
        h = _sha(text)
        if h in s["notes"] or len(s["notes"]) >= NOTES_PER_SESSION:
            return None
        result = _post("/api/v1/agent/session-note", {"session_id": session_hash, "client": CLIENT, "text": text}, NOTE_TIMEOUT_S)
        if isinstance(result, dict):
            s["notes"] = (s["notes"] + [h])[-NOTES_PER_SESSION:]
            log.debug("bibrain note saved=%s", result.get("saved"))
    except Exception as e:
        log.debug("bibrain note skipped: %s", type(e).__name__)
    return None


def register(ctx):
    """Called once by Hermes at startup (plugins guide). pre_llm_call carries the user's turn; post_llm_call carries the final reply."""
    ctx.register_hook("pre_llm_call", pre_llm_call)
    ctx.register_hook("post_llm_call", post_llm_call)
```

One difference from the CLI to carry on purpose: Python slices strings by code points, so `prompt[:PROMPT_MAX]` and `[:NOTE_MAX]` are the CLI's `firstCodePoints` already. Python's `len` is code points too, so `fits` matches `Array.from(s).length`.
